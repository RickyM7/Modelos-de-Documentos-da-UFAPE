\newcount\anoI \newcount\mesI \newcount\diaI

\def\lerinicio#1/#2/#3\relax{\diaI=#1 \mesI=#2 \anoI=#3 }
\def\lerfim#1/#2/#3\relax{\diaF=#1 \mesF=#2 \anoF=#3 }
\newcommand{\doisdig}[1]{\ifnum#1<10 0\fi\number#1}
\def\listaferiados{}
\newcommand{\calcbissexto}{%
  \bis=0 \auxc=\anoC \divide\auxc by 4 \multiply\auxc by 4
  \ifnum\auxc=\anoC \bis=1
    \auxc=\anoC \divide\auxc by 100 \multiply\auxc by 100
    \ifnum\auxc=\anoC \bis=0
      \auxc=\anoC \divide\auxc by 400 \multiply\auxc by 400
      \ifnum\auxc=\anoC \bis=1 \fi
    \fi
  \fi}
% Dia da semana de \diaC/\mesC/\anoC em \diasem (0 = domingo)
\newcommand{\calcsemana}{%
  \auxc=\anoC \ifnum\mesC<3 \advance\auxc by -1 \fi
  \diasem=\auxc
  \divide\auxc by 4 \advance\diasem by \auxc
  \auxc=\anoC \ifnum\mesC<3 \advance\auxc by -1 \fi
  \divide\auxc by 100 \advance\diasem by -\auxc
  \auxc=\anoC \ifnum\mesC<3 \advance\auxc by -1 \fi
  \divide\auxc by 400 \advance\diasem by \auxc
  \advance\diasem by \ifcase\mesC\or0\or3\or2\or5\or0\or3\or5\or1\or4\or6\or2\or4\fi\relax
  \advance\diasem by \diaC
  \auxc=\diasem \divide\auxc by 7 \multiply\auxc by 7 \advance\diasem by -\auxc}
\newcommand{\coletadias}{%
  \ifnum\diaC>\ultdia
  \else
    \calcsemana
    \ifnum\diasem>0
      \ifnum\diasem<6
        \edef\hoje{\doisdig\diaC/\doisdig\mesC/\number\anoC}%
        \expandafter\ifinlist\expandafter{\hoje}{\listaferiados}{}%
          {\ifdefempty{\dias}{\edef\dias{\doisdig\diaC}}{\edef\dias{\dias,\doisdig\diaC}}}%
      \fi
    \fi
    \advance\diaC by 1
    \expandafter\coletadias
  \fi}
\newcommand{\gerames}{%
  \calcbissexto
  \ultdia=\ifcase\mesC\or31\or\ifnum\bis=1 29\else 28\fi\or31\or30\or31\or30\or31\or31\or30\or31\or30\or31\fi\relax
  \diaC=1
  \ifnum\anoC=\anoI \ifnum\mesC=\mesI \diaC=\diaI \fi\fi
  \ifnum\anoC=\anoF \ifnum\mesC=\mesF \ultdia=\diaF \fi\fi
  \def\dias{}%
  \coletadias
  \ifdefempty{\dias}{}{%
    \edef\chamada{\noexpand\folha{\doisdig\mesC}{\number\anoC}{\dias}}%
    \chamada}%
  \advance\mesC by 1
  \ifnum\mesC>12 \mesC=1 \advance\anoC by 1 \fi}
\newcommand{\proximosmeses}{%
  \ifnum\anoC<\anoF \gerames \expandafter\proximosmeses
  \else\ifnum\anoC=\anoF \ifnum\mesC>\mesF \else \gerames \expandafter\expandafter\expandafter\proximosmeses\fi\fi\fi}
\newcommand{\gerarfolhas}{%
  \expandafter\lerinicio\FDataInicio\relax
  \expandafter\lerfim\FDataFim\relax
  \begingroup
  \def\lerferiado##1/##2/##3\relax{\xdef\feriado{\doisdig{##1}/\doisdig{##2}/\number##3}}%
  \def\do##1{\lerferiado##1\relax\listxadd\listaferiados{\feriado}}%
  \expandafter\docsvlist\expandafter{\FFeriados}%
  \endgroup
  \ifnum\numexpr(\anoF*12+\mesF)*32+\diaF\relax<\numexpr(\anoI*12+\mesI)*32+\diaI\relax
    \PackageError{ficha}{A data de fim é anterior à data de início}{Confira FDataInicio e FDataFim no bloco de dados.}%
  \fi%
  \anoC=\anoI \mesC=\mesI
  \proximosmeses}

% Guarda o último item da lista de dias (último dia de estágio do mês)
\newcommand{\salvaultimo}[1]{\def\ultimodia{#1}}

% Folha de um mês: preenche a referência, o rodapé e as linhas com as datas
\newcommand{\folha}[3]{%
  \begingroup
  \def\fmes{#1}\def\fano{#2}%
  \def\ultimodia{}%
  \forcsvlist{\salvaultimo}{#3}%
  \edef\FReferencia{\nomemes{#1} de #2}%
  \edef\FDia{\ultimodia}%
  \edef\FMes{\nomemes{#1}}%
  \edef\FAno{#2}%
  \renewcommand{\corpotabela}{\forcsvlist{\linhadia}{#3}}%
  \desenhafolha
  \endgroup
  \clearpage}
